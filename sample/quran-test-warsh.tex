% Typesetting the Quran in the riwayah of Warsh (compile with XeLaTeX).
% Warsh numbers verses differently from Hafs: 6214 verses in total.
\documentclass{article}
\usepackage[a5paper,margin=1.5cm]{geometry}
\usepackage{fontspec}
\usepackage{arabxetex}
% Font: "KFGQPC WARSH Uthmanic Script" version 0.10 (warsh.10.ttf) by the King Fahd Glorious
% Quran Printing Complex (KFGQPC). Its end-user license allows free use, but forbids selling
% or modifying it, so it is NOT distributed with this package. Download exactly this version:
%   https://raw.githubusercontent.com/thetruetruth/quran-data-kfgqpc/main/warsh/font/warsh.10.ttf
% (a mirror of the KFGQPC font; sha256 e6347a1b4cf0ca32...). The newer versions on the KFGQPC
% site (https://fonts.qurancomplex.gov.sa/warsh-reading, V2.1 and V3.0) encode the text
% differently and do not match this text yet.
% Install it (e.g. in ~/.fonts, then run fc-cache), or place warsh.10.ttf next to this file
% and use instead:
%   \newfontfamily\arabicfont[Script=Arabic,Path=./,Extension=.ttf]{warsh.10}
\newfontfamily\arabicfont[Script=Arabic]{KFGQPC WARSH Uthmanic Script}
% Fallback that covers every character of the text (Hafs/Mashriqi letter forms):
% \newfontfamily\arabicfont[Script=Arabic]{Amiri Quran}
\usepackage[warsh]{quran}

% The Warsh font draws the verse number inside its own ornament, so the braces of the
% package around the number are blanked.
\SetAyahMarkerStyle{}{}

\begin{document}
\parindent=0pt
\thispagestyle{empty}

\section*{Surah \surahname[1]}
\begin{arab}[utf]
  \quransurah[1]
\end{arab}

\section*{Last verse of Al-Baqarah (2:285 in Warsh numbering)}
\begin{arab}[utf]
  \quranayah[2][285]
\end{arab}

\section*{Juz 30, first surah}
\begin{arab}[utf]
  \quransurah*[78]
\end{arab}
\end{document}
