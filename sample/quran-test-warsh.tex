% Typesetting the Quran in the riwayah of Warsh (compile with XeLaTeX).
% Warsh numbers verses differently from Hafs: 6214 verses in total.
\documentclass{article}
\usepackage[a5paper,margin=1.5cm]{geometry}
\usepackage{fontspec}
\usepackage{arabxetex}
% Font: "KFGQPC Warsh V3" (version 3.0) by the King Fahd Glorious Quran Printing Complex
% (KFGQPC). Its end-user license allows free use, but forbids selling or modifying it, so it
% is NOT distributed with this package. Download it (zip with the font and a Word file):
%   page:   https://fonts.qurancomplex.gov.sa/warsh-reading
%   direct: https://fonts.qurancomplex.gov.sa/?fdc_download=https%3A%2F%2Ffonts.qurancomplex.gov.sa%2Ffonts%2FKFGQPC-Warsh-V30.zip
% The text of the package is the one of that release; older versions of the font (V2.1 and
% earlier) lack some of its characters. Install it (e.g. in ~/.fonts, then run fc-cache), or
% place "KFGQPC Warsh V30.ttf" next to this file and use instead:
%   \newfontfamily\arabicfont[Script=Arabic,Path=./,Extension=.ttf]{KFGQPC Warsh V30}
\newfontfamily\arabicfont[Script=Arabic]{KFGQPC Warsh V3}
% Other fonts (Amiri Quran, Scheherazade) lack some characters of this text.
\usepackage[warsh]{quran}
\SetAyahMarkerStyle{۝}{}

\begin{document}
\parindent=0pt
\thispagestyle{empty}

\section*{Surah \surahname[1]}
\begin{arab}[utf]
  \quransurah[1]
\end{arab}

\section*{Last verse of Al-Baqarah (2:285 in Warsh numbering)}
\begin{arab}[utf]
  \quranayah[2][285]
\end{arab}

\section*{Juz 30, first surah}
\begin{arab}[utf]
  \quransurah*[78]
\end{arab}
\end{document}
